// Lambda 表达式
lambda
    : FUNC ID* COLON block            # LambdaWithBlock
    | FUNC ID* expression             # LambdaWithExpr
    | ID COLON block                  # LambdaIdWithBlock    // 处理"函参数"被识别为ID的情况
    | ID expression                   # LambdaIdWithExpr     // 处理"函参数"被识别为ID的情况
    ;
